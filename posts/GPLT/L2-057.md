## 思路
用两个数组存储数据，第一个数组存储已经通过的作业（模拟栈），第二个数组存储还未通过的作业

注意，用临时数组存储之后赋值给第二个数组之前，需要反转（想象一下最先批的作业叠在最下面）

## 代码
```c++
#include<bits/stdc++.h>
using namespace std;
int n,T;
int main(){
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cin>>n>>T;
	vector<int>ans;
	vector<pair<int,int>>path(n);
	for(int i=0;i<n;i++){
		cin>>path[i].first;
		path[i].second=i+1;
	}
	while(ans.size()!=n){
		vector<pair<int,int>>tem;
		for(auto &t:path){
			if(t.first<=T){
				ans.push_back(t.second);
			}
			else {
				tem.push_back(t);
			}
		}
		if(tem.empty()) break;
		reverse(tem.begin(),tem.end()); 
		path=tem;
		int sum=0;
		for(auto &t:path){
			sum+=t.first;
		} 
		sum/=path.size();
		T=sum;
	}
	for(int i=0;i<ans.size();i++){
		if(i)cout<<" ";
		cout<<ans[i];
	}
	return 0;
}
```
