## 题目解读
题目说的非常复杂，这里可以画成一棵树来解释

![1](/images/L2-059.jpg)


## 思路
用vector<vector<pair<int,int>>>gra(n);来存储数据，用bool数组来记录是否为叶子节点

遍历树中的每一条路，找出每条路上的系数的最小值

最后，找出叶子节点路径中的最大值

## 代码
```c++
#include<bits/stdc++.h>
using namespace std;
int n;
int main(){
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cin>>n;
	//处理输入，邻接表建树，用bool数组记录是否为子节点 
	vector<vector<pair<int,int>>>gra(n);
	vector<bool>father(n);
	for(int i=1;i<n;i++){
		int b,val;
		cin>>b>>val;
		gra[b].push_back({i,val});
		father[b]=true;
	}
	//遍历树（BFS/DFS均可，这里用BFS），记录每个节点路径最小值 
	vector<int>dis(n,105);
	queue<int>q;
	q.push(0);
	while(!q.empty()){
		int t=q.front();
		q.pop();
		for(auto &k:gra[t]){
			dis[k.first]=min(dis[t],k.second);//路径上的最小安全系数的传递
			q.push(k.first);
		}
	}
	//处理输出，记录子节点路径最大值 
	int ans=-1;
	for(int i=1;i<n;i++){
		if(!father[i]){
			ans=max(ans,dis[i]);
		}
	}
	cout<<ans<<'\n'; 
	vector<int>s;
	for(int i=1;i<n;i++){
		if(!father[i]&&dis[i]==ans){
			s.push_back(i);
		}
	}
	for(int i=0;i<s.size();i++){
		if(i)cout<<" ";
		cout<<s[i];
	}
	return 0;
}
```
