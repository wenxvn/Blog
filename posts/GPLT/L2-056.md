## 思路一

一开始想的是直接遍历检测，可以拿14/25分

## 代码
```c++
#include<bits/stdc++.h>
using namespace std;
#define int long long
int n,a,b;
vector<int>ans;
bool solve(int x){
	string s=to_string(x);
	if(s.size()!=n)return false;
	int t=0;
	for(int i=0;i<n;i++){
		t=t*10+(s[i]-'0');
		int len=i+1;
		if(t%len!=0)return false;
	}
	return true;
}
signed main(){
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cin>>n>>a>>b;
	for(int i=a;i<=b;i++){
		if(solve(i))ans.push_back(i);
	}
	if(ans.empty())cout<<"No Solution";
	else {
		for(int i=0;i<ans.size();i++){
			cout<<ans[i]<<'\n';
		}
	}
	return 0;
}
```

## 优化思路

[a,b]数据比较多，但符合要求的数据却比较少，可以考虑用DFS搜索

## 代码

```c++
#include<bits/stdc++.h>
using namespace std;
#define int long long
int n,a,b;
vector<int>ans;
void dfs(int pos,int t){
	if(pos==n){
		if(t>=a&&t<=b)ans.push_back(t);
		return ;
	}
	for(int i=0;i<=9;i++){
		int next=t*10+i; 
		if(next%(pos+1)==0)dfs(pos+1,next);
	}
}
signed main(){
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cin>>n>>a>>b;
	for(int i=1;i<=9;i++){
		dfs(1,i);
	}
	if(ans.empty())cout<<"No Solution";
	else {
		for(int i=0;i<ans.size();i++){
			cout<<ans[i]<<'\n';
		}
	}
	return 0;
}
```
