## 思路
如何存数据？

有向图，用vector<vector<pair<int,int>>>gra(n+1);邻接表存。第0项存出发点，第1项(pair项)的first存到达点，second存概率

模拟一下思维路径，开始设置bool数组来判断这个节点是否被访问过

如果没访问过，判断一下最大概率和最小下标

如果遍历了一遍还是没找到未访问的，就代表已经没路了，退出循环

## 代码
```c++
#include<bits/stdc++.h>
using namespace std;
int n,m;
int main(){
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cin>>n>>m;
	vector<vector<pair<int,int>>>gra(n+1);
	int a,b,val;
	for(int i=0;i<m;i++){
		cin>>a>>b>>val;
		gra[a].push_back({b,val});
	}
	int t;
	cin>>t;
	while(t--){
		int k;
		cin>>k;
		cout<<k;
		vector<bool>vis(n+1);
		vis[k]=true;
		while(1){
			int index=-1;
			int maxval=0;
			for(auto g:gra[k]){
				if(!vis[g.first]){
					if(g.second>maxval){
						maxval=g.second;
						index=g.first;
					}
					else if(g.second==maxval&&index>g.first){
						index=g.first;
					}
				}
			}
			if(index==-1)break;
			vis[index]=true;
			k=index;
			cout<<"->"<<index;
		}
		cout<<'\n';
	}
	return 0;
}
```

当然，如果不在每次选节点时比较大小，而是一开始就用sort排好序，也是可以的

```c++
#include<bits/stdc++.h>
using namespace std;
int n,m;
bool com(const pair<int,int>& x,const pair<int,int>& y){
	if(x.second!=y.second)return x.second>y.second;
	else return x.first<y.first;
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cin>>n>>m;
	vector<vector<pair<int,int>>>gra(n+1);
	int a,b,val;
	for(int i=0;i<m;i++){
		cin>>a>>b>>val;
		gra[a].push_back({b,val});
	}
	for(int i=1;i<=n;i++){
		sort(gra[i].begin(),gra[i].end(),com);
	}
	int t;
	cin>>t;
	while(t--){
		int k;
		cin>>k;
		cout<<k;
		vector<bool>vis(n+1);
		vis[k]=true;
		while(1){
			int index=-1;
			for(auto g:gra[k]){
				if(!vis[g.first]){
					index=g.first;
					break;
				}
			}
			if(index==-1)break;
			vis[index]=true;
			k=index;
			cout<<"->"<<index;
		}
		cout<<'\n';
	}
	return 0;
}
```
