# MC0505 厨房里练手艺

```c++
#include<bits/stdc++.h>
using namespace std;
int n,q;
int main(){
 ios::sync_with_stdio(false);
 cin.tie(nullptr);
 cin>>n>>q;
 vector<int>ans(n+1);
 for(int i=1;i<=n;i++)cin>>ans[i];
 while(q--){
  int k,l,r;cin>>k>>l>>r;
  if(k==1){
   int x;cin>>x;
   for(int i=l;i<=r;i++){
    ans[i]+=x;
   }
  }
  else if(k==2){
   int x;cin>>x;
   for(int i=l;i<=r;i++){
    if(ans[i]>x)ans[i]=x;
   }
  }
  else {
   int sum=0;
   for(int i=l;i<=r;i++){
    sum+=ans[i];
   }
   cout<<sum<<'\n';
  }
 }
 return 0;
}
```

# MC0506 摆酒席设圈套
## 思路

注意是任意符合条件的四个位置，不是连续的四个位置

预处理：枚举前两项符合要求的个数，枚举后两项符合要求的个数

最后，枚举中间两项符合要求的个数

公式为 ans = (ans + L[w] * R[e]) % MOD;

如果中间两项符合要求，那么前两项符合的个数，和后两项符合的个数，排列组合一下

## 代码
```c++
#include<bits/stdc++.h>
using namespace std;
#define int long long
int n,cnt;
signed main(){
 ios::sync_with_stdio(false);
 cin.tie(nullptr);
 cin>>n;
 vector<int>a(n);
 for(int i=0;i<n;i++)cin>>a[i];
 vector<int>left(n);
 vector<int>right(n);
 for(int i=0;i<n;i++){
  for(int j=0;j<i;j++){
   if(a[j]<a[i])left[i]++;
  }
 }
 for(int i=0;i<n;i++){
  for(int j=i+1;j<n;j++){
   if(a[i]>a[j])right[i]++;
  }
 }
 for(int i=0;i<n;i++){
  for(int j=i+1;j<n;j++){
   if(a[i]==a[j])cnt=(cnt+left[i]*right[j])%10007;
  }
 }
 cout<<cnt;
 return 0;
}
```
