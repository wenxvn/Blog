## 思路

DFS模板题

注意剪枝：

1、总和超过n或总和等于n但没存满10个数

2、n>30或n<10时不满足条件

## 代码
```c++
#include<bits/stdc++.h>
using namespace std;
int n;
int cnt;
vector<vector<int>>ans;
vector<int>path;
void dfs(int x,int sum){
	if(sum>n||sum==n&&x<10)return ;
	if(x==10&&sum==n){
		cnt++;
		ans.push_back(path);
	}
	for(int i=1;i<=3;i++){
		path.push_back(i);
		dfs(x+1,sum+i);
		path.pop_back();
	}
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cin>>n;
	if(n>30||n<10){
		cout<<0;
		return 0;
	}
	dfs(0,0);
	cout<<cnt<<'\n';
	for(int i=0;i<ans.size();i++){
		for(int j=0;j<10;j++){
			cout<<ans[i][j]<<" ";
		}
		cout<<'\n';
	}
	return 0;
}
```
